  STEP 4 — ENABLE GITHUB PAGES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. Repo → Settings → Pages  (left sidebar)
  2. Source: main branch, / (root) → Save
  3. Custom domain: fim-designs.com → Save
  4. Don't enable HTTPS yet — do that after DNS in Step 5

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  STEP 5 — CONFIGURE GODADDY DNS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  godaddy.com → My Products → fim-designs.com → DNS

  DELETE any existing A record for "@"

  ADD these 4 A records:
  ┌──────┬──────┬─────────────────────┬─────┐
  │ Type │ Name │ Value               │ TTL │
  ├──────┼──────┼─────────────────────┼─────┤
  │ A    │  @   │ 185.199.108.153     │ 600 │
  │ A    │  @   │ 185.199.109.153     │ 600 │
  │ A    │  @   │ 185.199.110.153     │ 600 │
  │ A    │  @   │ 185.199.111.153     │ 600 │
  └──────┴──────┴─────────────────────┴─────┘

  ADD this CNAME record:
  ┌───────┬──────┬────────────────────────────────┬──────┐
  │ Type  │ Name │ Value                          │ TTL  │
  ├───────┼──────┼────────────────────────────────┼──────┤
  │ CNAME │ www  │ YOUR-USERNAME.github.io        │ 3600 │
  └───────┴──────┴────────────────────────────────┴──────┘

  Example: GitHub username = faysalim → value = faysalim.github.io

  ⏳  Wait 30-60 minutes (up to 48h). Check: dnschecker.org

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  STEP 6 — ENABLE HTTPS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Repo → Settings → Pages → tick "Enforce HTTPS" → Save
  (Available after DNS propagates. GitHub provides SSL free.)
